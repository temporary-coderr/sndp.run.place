name: YouTube Auto Bot - FIXED NO CACHE ERROR

on:
  schedule:
    - cron: '0 0 * * *'
  workflow_dispatch:

jobs:
  bot:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Setup Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install libraries
        run: |
          pip install edge-tts==6.1.10 moviepy==1.0.3 Pillow==10.4.0 requests==2.31.0 gtts==2.5.1 google-api-python-client==2.143.0 google-auth-oauthlib==1.2.1
          sudo apt-get update && sudo apt-get install -y ffmpeg
          mkdir -p clips

      - name: Restore YouTube token
        env:
          YOUTUBE_TOKEN_JSON: ${{ secrets.YOUTUBE_TOKEN_JSON }}
        run: echo "$YOUTUBE_TOKEN_JSON" > token.json

      - name: Generate video
        env:
          GROQ_API_KEY: ${{ secrets.GROQ_API_KEY }}
          PIXABAY_API_KEY: ${{ secrets.PIXABAY_API_KEY }}
        run: |
          timeout 600 python final_google_free_bot.py || echo "Video generation timed out after 10 min, using fallback"

      - name: Upload to YOUR channel
        run: python youtube_uploader.py

      - name: Upload artifact
        uses: actions/upload-artifact@v4
        with:
          name: video-${{ github.run_number }}
          path: video_*.mp4
          retention-days: 7
